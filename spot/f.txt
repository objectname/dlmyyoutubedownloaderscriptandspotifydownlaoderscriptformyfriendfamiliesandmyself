Get-ChildItem -Path "spot/Downloads" -Recurse | ForEach-Object { $d = $_.FullName.Replace((Resolve-Path "spot/Downloads").Path, (New-Item -ItemType Directory -Path "spot/Downloads-mp3" -Force).FullName); if ($_.PSIsContainer) { New-Item -ItemType Directory -Path $d -Force > $null } elseif ($_.Extension -eq ".m4a") { $out = [System.IO.Path]::ChangeExtension($d, ".mp3"); ffmpeg -i $_.FullName -c:a libmp3lame -q:a 2 $out -y } else { Copy-Item -Path $_.FullName -Destination $d -Force } }
